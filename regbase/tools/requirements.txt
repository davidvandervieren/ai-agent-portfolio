# RegBase toolchain.
#
#     python3 -m pip install -r regbase/tools/requirements.txt      # Linux/macOS
#     py -m pip install -r regbase\tools\requirements.txt           # Windows
#
# Version floors, not exact pins. Exact pins were the original choice and they
# broke immediately: PyYAML==6.0.1 has no wheel for CPython 3.14, so pip fell
# back to compiling the C extension and failed with "Microsoft Visual C++ 14.0
# or greater is required" on a machine that had no compiler and had no reason
# to need one. Floors let pip pick the newest wheel built for whatever Python
# is present, which is what this toolchain actually needs — it has no API
# surface tight enough to justify pinning, and it gets run on whatever laptop
# is handy.
#
# NOTE (Debian/Ubuntu): an externally-managed Python needs
#     python3 -m pip install --break-system-packages -r ...
#
# NOTE (Debian images): a mismatched cffi/cryptography pair surfaces as
#   "ImportError: ... _cffi_backend", or an AttributeError deep in requests'
#   TLS stack the moment harvest.py opens its first HTTPS connection. Repair:
#     pip install --break-system-packages --upgrade cffi cryptography
#   (cryptography vendors its own OpenSSL; do not downgrade system OpenSSL.)

# registry + schema
PyYAML>=6.0.2                # sources/*.yaml, people/*.yaml           (common.py)
                             #   6.0.2 is the floor that matters: it is the
                             #   first release with cp313/cp314 wheels.
jsonschema>=4.23             # schemas/*.json validation               (common.validate_sources)

# fetching
requests>=2.32               # polite HTTP with conditional GET        (common.Fetcher)

# text extraction
beautifulsoup4>=4.12         # HTML -> text/headings                   (extract.py)
lxml>=5.2                    # fast parser backend for bs4 + KML/XML   (extract.py, kmlgeo.py)
pypdf>=4.2                   # PDF text extraction                     (extract.py)
pdfminer.six>=20231228       # fallback PDF layout extraction          (extract.py)

# geometry
# Only needed by the optional projection paths; kmlgeo.py's core length and
# area math is pure stdlib on purpose, so the KML pipeline still runs if these
# two refuse to install.
shapely>=2.0                 # geometry ops, buffers, intersections    (kmlgeo.py)
pyproj>=3.6                  # CRS transforms (WGS84 <-> state plane)  (kmlgeo.py)

# browser rendering (optional, but required for Municode / eCode360 /
# Franklin Legal, which deliver their ordinances over XHR and harvest as empty
# shells without it)
#     py -m pip install playwright
# On Windows the renderer uses the Chrome or Edge you already have, so
# `playwright install` is usually unnecessary.
playwright>=1.40
